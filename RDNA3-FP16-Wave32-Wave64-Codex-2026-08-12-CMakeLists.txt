# 해당코드는 Codex로 수정됨
cmake_minimum_required(VERSION 3.25)
project(rdna3_pure_fp16_wave LANGUAGES CXX)

find_package(Vulkan 1.4.344 REQUIRED)

add_library(rdna3_fp16_wave STATIC
    host-rdna3_fp16_wave_pipeline.cpp
)

target_include_directories(rdna3_fp16_wave
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
)

target_compile_features(rdna3_fp16_wave PUBLIC cxx_std_20)
target_link_libraries(rdna3_fp16_wave PUBLIC Vulkan::Vulkan)

if(MSVC)
    target_compile_options(rdna3_fp16_wave PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(rdna3_fp16_wave PRIVATE
        -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
endif()
